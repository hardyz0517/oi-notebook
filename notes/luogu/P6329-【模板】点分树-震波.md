#include <bits/stdc++.h>
using namespace std;

const int N = 1e5 + 5;

int n, m, val[N];
vector<int> g[N];

int pa[N], dep[N], sz[N], son[N], top[N];
void dfs1(int u, int p)
{
    pa[u] = p;
    dep[u] = dep[p] + 1;
    sz[u] = 1;
    for (int v : g[u])
    {
        if (v == p)
            continue;
        dfs1(v, u);
        sz[u] += sz[v];
        if (sz[v] > sz[son[u]])
            son[u] = v;
    }
}
void dfs2(int u, int t)
{
    top[u] = t;
    if (son[u])
        dfs2(son[u], t);
    else
        return;
    for (int v : g[u])
    {
        if (v == pa[u] or v == son[u])
            continue;
        dfs2(v, v);
    }
}
int get_lca(int u, int v)
{
    while (top[u] != top[v])
    {
        if (dep[top[u]] < dep[top[v]])
            swap(u, v);
        u = pa[top[u]];
    }
    return dep[u] < dep[v] ? u : v;
}
int get_dist(int u, int v)
{
    return dep[u] + dep[v] - 2 * dep[get_lca(u, v)];
}

int siz[N], mx[N], rt;
bool del[N];
int fa[N];
vector<int> tr1[N], tr2[N];

void get_siz(int u, int p)
{
    siz[u] = 1;
    for (int v : g[u])
    {
        if (v == p or del[v])
            continue;
        get_siz(v, u);
        siz[u] += siz[v];
    }
}
void get_rt(int u, int p, const int tot)
{
    siz[u] = 1;
    mx[u] = 0;
    for (int v : g[u])
    {
        if (v == p or del[v])
            continue;
        get_rt(v, u, tot);
        siz[u] += siz[v];
        mx[u] = max(mx[u], siz[v]);
    }
    mx[u] = max(mx[u], tot - siz[u]);
    if (mx[u] < mx[rt])
        rt = u; // rt[0]=2e9
}
void get_max_dist(int x, int p, int d, int &md)
{
    md = max(md, d);
    for (int y : g[x])
    {
        if (y == p or del[y])
            continue;
        get_max_dist(y, x, d + 1, md);
    }
}

void get_max_dist_fa(int x, int p, int tar, int &md)
{
    md = max(md, get_dist(x, tar));
    for (int y : g[x])
    {
        if (y == p or del[y])
            continue;
        get_max_dist_fa(y, x, tar, md);
    }
} // 连通块内点到 target (父重心) 的最大距离
void build(int u)
{
    del[u] = 1;
    int md1 = 0;
    get_max_dist(u, 0, 0, md1);
    tr1[u].assign(md1 + 2, 0);

    if (fa[u])
    {
        int md2 = 0;
        get_max_dist_fa(u, 0, fa[u], md2);
        tr2[u].assign(md2 + 2, 0);
    }

    for (int v : g[u])
    {
        if (del[v])
            continue;

        get_siz(v, 0);
        rt = 0;
        mx[0] = 2e9;
        get_rt(v, 0, siz[v]);

        fa[rt] = u;
        build(rt);
    }
}
void add(vector<int> &t, int d, int v)
{
    d++; // 防止触碰0出bug
    for (; d < t.size(); d += d & -d)
        t[d] += v;
}
int query(vector<int> &t, int d)
{
    int res = 0;
    d++;
    d = min((int)t.size() - 1, d);
    for (; d; d -= d & -d)
        res += t[d];
    return res;
}
void update(int u, int w)
{
    int dt = w - val[u];
    val[u] = w;
    add(tr1[u], 0, dt);

    for (int curr = u; fa[curr]; curr = fa[curr])
    {
        int p = fa[curr];
        int d = get_dist(u, p);

        add(tr1[p], d, dt);
        add(tr2[curr], d, dt); // 我的退款账本
    }
}
// dist(p,v)<=k-dist(u,p)

int solve(int u, int k)
{
    int res = query(tr1[u], k);
    int curr = u;
    while (fa[curr] != 0)
    {
        int p = fa[curr];
        int d = get_dist(u, p);
        if (k >= d)
        {
            res += query(tr1[p], k - d);
            res -= query(tr2[curr], k - d);
        }
        curr = p;
    }
    return res;
} // p会一直往上跳只要辐射范围够大
int main()
{
    cin.tie(0)->sync_with_stdio(0);

    cin >> n >> m;
    for (int i = 1; i <= n; i++)
        cin >> val[i];
    for (int i = 1; i < n; i++)
    {
        int u, v;
        cin >> u >> v;
        g[u].push_back(v);
        g[v].push_back(u);
    }

    dfs1(1, 0);
    dfs2(1, 1);

    get_siz(1, 0);
    mx[0] = 2e9;
    rt = 0;
    get_rt(1, 0, siz[1]);

    build(rt);

    for (int i = 1; i <= n; i++)
    {
        int w = val[i];
        val[i] = 0;
        update(i, w);
    }

    int ans = 0;
    while (m--)
    {
        int op, x, y;
        cin >> op >> x >> y;
        x ^= ans;
        y ^= ans;
        if (op == 0)
        {
            ans = solve(x, y);
            cout << ans << "\n";
        }
        else
            update(x, y);
    }
    return 0;
}
